Async: Callbacks & Promises

JavaScript is single-threaded but non-blocking. Understanding how async work is scheduled — and how to express it cleanly — is the core of modern JS.

Topic 1: Sync vs Async

Synchronous code runs top-to-bottom with no gaps. Asynchronous code schedules work for later — the engine keeps going and comes back when the task is ready.

// SYNCHRONOUS — runs in order, no surprises
console.log("1");
console.log("2");
console.log("3");
// Output: 1, 2, 3

// ASYNCHRONOUS — setTimeout schedules; doesn't pause
console.log("1");
setTimeout(() => console.log("2"), 0);   // goes through the event loop
console.log("3");
// Output: 1, 3, 2
// Even with delay 0! JS finishes synchronous code first; async waits its turn.

Topic 2: Callbacks

A callback is a function you pass to be called "when done". Node-style callbacks use the convention (err, data) — the first argument signals failure, the second carries the result.

function fetchUser(id, callback) {
  console.log(`Fetching user ${id}...`);
  setTimeout(() => {
    const user = { id, name: "Priya" };
    callback(null, user);   // null = no error
  }, 1000);
}

fetchUser(7, (err, user) => {
  if (err) { console.error("Failed:", err); return; }
  console.log("Got user:", user);
});
// Logs:
//   Fetching user 7...
//   (1 second later)
//   Got user: { id: 7, name: "Priya" }

Topic 3: Callback Hell

Nesting callbacks to sequence async steps creates the "pyramid of doom" — indentation explodes, error handling repeats at every level, and adding steps means rewriting the structure.

// The "pyramid of doom"
fetchUser(7, (err, user) => {
  if (err) { console.error(err); return; }
  fetchOrders(user.id, (err, orders) => {
    if (err) { console.error(err); return; }
    fetchItems(orders[0].id, (err, items) => {
      if (err) { console.error(err); return; }
      console.log(items);   // finally got there!
    });
  });
});
// Problems: indentation explodes, error handling repeated,
// hard to add steps, impossible to return values.

Topic 4: Promise States

A Promise is an object representing a future value. It starts pending and settles exactly once — to fulfilled (with a result) or rejected (with an error).

// Create a promise
const p = new Promise((resolve, reject) => {
  setTimeout(() => {
    const success = true;
    if (success) resolve("Done!");          // → fulfilled
    else         reject(new Error("Oops")); // → rejected
  }, 1000);
});
// p is PENDING. After 1s it settles.

// Real-world pattern: wrapping setTimeout
function delay(ms) {
  return new Promise((resolve) => setTimeout(resolve, ms));
}

delay(500).then(() => console.log("half a second passed"));

Topic 5: .then / .catch / .finally

Chain .then calls to sequence steps without nesting. Return a value from .then to pass it forward; return a Promise to make the chain wait for it. .catch handles any failure in the chain; .finally runs regardless.

function fetchUser(id) {
  return new Promise((resolve, reject) => {
    setTimeout(() => {
      if (id < 0) reject(new Error("Bad id"));
      else resolve({ id, name: "Priya" });
    }, 500);
  });
}

fetchUser(7)
  .then((user) => {
    console.log("got user:", user);
    return user.id;              // plain value → next .then gets it
  })
  .then((id) => {
    return fetchUser(id + 1);   // returning a Promise → chain awaits it
  })
  .then((nextUser) => {
    console.log("next user:", nextUser);
  })
  .catch((err) => {
    console.error("any failure caught here:", err.message);
  })
  .finally(() => {
    console.log("done — runs whether success or fail");
  });

Topic 6: Promise Combinators

Run multiple Promises in parallel rather than sequencing them one-by-one.

const p1 = fetch("/user");
const p2 = fetch("/orders");
const p3 = fetch("/items");

// all — wait for everything; fail-fast if any rejects
Promise.all([p1, p2, p3])
  .then(([user, orders, items]) => console.log("all done"))
  .catch((err) => console.error("at least one failed:", err));

// allSettled — wait for everything, never throws
Promise.allSettled([p1, p2, p3])
  .then((results) => {
    results.forEach((r) => {
      if (r.status === "fulfilled") console.log("ok:", r.value);
      else                          console.log("fail:", r.reason);
    });
  });

// race — first to settle wins (success OR failure)
Promise.race([p1, delay(5000)])    // timeout pattern
  .then((winner) => console.log("first done:", winner));

// any — first to FULFILL wins; all-reject → AggregateError
Promise.any([p1, p2, p3])
  .then((firstSuccess) => console.log("got one:", firstSuccess))
  .catch((err) => console.error("everyone failed"));

Topic 7: Promise.resolve / Promise.reject

Create already-settled Promises instantly — useful for caching, testing, or normalising a function that might return either a value or a Promise.

// Already-fulfilled
const cached = Promise.resolve({ id: 1, name: "Priya" });
cached.then((u) => console.log(u));   // logs immediately

// Already-rejected
const failed = Promise.reject(new Error("nope"));
failed.catch((e) => console.log("caught:", e.message));

// Common pattern: cache-or-fetch
function getUser(id, cache) {
  if (cache[id]) return Promise.resolve(cache[id]);  // instant
  return fetchUser(id);                               // real async
}
// Caller doesn't care which path — both return a Promise
getUser(7, {}).then(console.log);
