Task 1: Leading-edge debounce

Standard debounce fires after silence. Write a debounceLeading(fn, delay) that fires immediately on the first call of a burst, then ignores subsequent calls until quiet. Useful for submit buttons — the user gets instant feedback but can't double-click.

function debounceLeading(fn, delay) {
  let timer = null;
  return function (...args) {
    if (timer === null) {
      fn.apply(this, args);   // fire immediately on first call
    }
    clearTimeout(timer);
    timer = setTimeout(() => { timer = null; }, delay);
  };
}

const submit = debounceLeading((label) => {
  console.log("submit →", label);
}, 300);

[1, 2, 3, 4].forEach((n) => submit(`call-${n}`));
setTimeout(() => submit("call-5"), 700);
setTimeout(() => submit("call-6"), 750);

// Expected:
//   submit → call-1    (first call of burst 1 — fires immediately)
//   submit → call-5    (first call of burst 2 — fires immediately)

Task 2: memoize

Write a memoize(fn) wrapper that caches results by argument signature. Identical calls should return the cached value without running fn again.

function memoize(fn) {
  const cache = new Map();
  return function (...args) {
    const key = JSON.stringify(args);
    if (cache.has(key)) {
      console.log("cache hit →", key);
      return cache.get(key);
    }
    const result = fn.apply(this, args);
    cache.set(key, result);
    return result;
  };
}

function slowSquare(n) { return n * n; }
const fastSquare = memoize(slowSquare);

console.log(fastSquare(4));   // computed: 16
console.log(fastSquare(4));   // cache hit → [4], returns 16
console.log(fastSquare(7));   // computed: 49
console.log(fastSquare(7));   // cache hit → [7], returns 49

Task 3: Safe innerHTML wrapper

Write setHTML(element, html) that strips <script> blocks and on* event attributes before calling innerHTML. This is a minimal sanitiser — DOMPurify is the production choice.

function setHTML(element, html) {
  let safe = html.replace(/<script[\s\S]*?<\/script>/gi, "");
  safe = safe.replace(/\s+on\w+\s*=\s*["'][^"']*["']/gi, "");
  element.innerHTML = safe;
}

const div = document.createElement("div");
setHTML(div, '<p>Hello</p><script>alert("xss")<\/script><img src=x onerror="alert(1)">');
console.log(div.innerHTML);
// Expected: <p>Hello</p><img src="x">  (script gone, onerror gone)

Task 4: Benchmark with performance.now()

Measure the difference between building a large string with += concatenation versus Array.join(). Use performance.now() to time both and log the results.

function buildStringConcat(n) {
  let s = "";
  for (let i = 0; i < n; i++) s += "x";
  return s;
}

function buildStringJoin(n) {
  const arr = [];
  for (let i = 0; i < n; i++) arr.push("x");
  return arr.join("");
}

const N = 100_000;
const t0 = performance.now();
buildStringConcat(N);
const t1 = performance.now();
buildStringJoin(N);
const t2 = performance.now();

console.log(`concat: ${(t1 - t0).toFixed(3)} ms`);
console.log(`join  : ${(t2 - t1).toFixed(3)} ms`);
// join tends to be faster for large N because V8 can pre-allocate the buffer.
