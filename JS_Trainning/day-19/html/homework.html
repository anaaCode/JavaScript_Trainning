Exercise 1: Convert .then Chains

Convert ALL the .then chains from Day 18's homework into async/await with try/catch. Each exercise should become a named async function.

// Example: Day 18 Exercise 2 rewritten as async/await
function fetchData(id) {
  return new Promise((resolve, reject) => {
    setTimeout(() => {
      if (Math.random() > 0.5) resolve({ id, data: "payload" });
      else reject(new Error(`fetchData(${id}) failed`));
    }, 300);
  });
}

async function runChain() {
  try {
    const r1 = await fetchData(1);
    console.log("step 1:", r1);
    const r2 = await fetchData(2);
    console.log("step 2:", r2);
    const r3 = await fetchData(3);
    console.log("step 3:", r3);
  } catch (err) {
    console.error("caught:", err.message);
  }
}

runChain();

Exercise 2: fetchAllUsers in Parallel

Write fetchAllUsers(ids) that takes an array of IDs and returns all users in parallel using Promise.all + map. Time it against a sequential version and confirm the parallel version is faster.

function fetchUser(id) {
  return new Promise((resolve) =>
    setTimeout(() => resolve({ id, name: `User ${id}` }), 500)
  );
}

// Parallel version — all fetches fire at once
async function fetchAllUsers(ids) {
  const t0 = Date.now();
  const users = await Promise.all(ids.map((id) => fetchUser(id)));
  console.log("parallel:", Date.now() - t0, "ms");  // ~500ms
  return users;
}

// Sequential version — fetches one at a time
async function fetchAllUsersSeq(ids) {
  const t0 = Date.now();
  const users = [];
  for (const id of ids) {
    users.push(await fetchUser(id));
  }
  console.log("sequential:", Date.now() - t0, "ms");  // ~2000ms for 4 ids
  return users;
}

fetchAllUsers([1, 2, 3, 4]).then(console.log);
fetchAllUsersSeq([1, 2, 3, 4]).then(console.log);

Exercise 3: withTimeout Helper

Write withTimeout(promise, ms) that rejects with "timeout" if the original Promise doesn't settle within ms. Use Promise.race internally.

function withTimeout(promise, ms) {
  const timer = new Promise((_, reject) =>
    setTimeout(() => reject(new Error("timeout")), ms)
  );
  return Promise.race([promise, timer]);
}

// Test — fetchUser takes 1000ms, timeout is 500ms
function fetchUser(id) {
  return new Promise((resolve) =>
    setTimeout(() => resolve({ id, name: "Priya" }), 1000)
  );
}

async function run() {
  try {
    const user = await withTimeout(fetchUser(7), 500);
    console.log("got user:", user);
  } catch (err) {
    console.error("error:", err.message);   // "timeout"
  }
}

run();

Exercise 4: Timed for...of Loop

Write an async function that loops over [300, 100, 200] with for...of, waits each amount in ms, and logs when each finishes. Confirm the order matches the input array, not the delay size.

function wait(ms) {
  return new Promise((resolve) => setTimeout(resolve, ms));
}

async function timedLoop() {
  const delays = [300, 100, 200];
  for (const ms of delays) {
    await wait(ms);
    console.log(`waited ${ms}ms`);
  }
  console.log("all done");
}

timedLoop();
// Output (in order):
//   waited 300ms
//   waited 100ms
//   waited 200ms
//   all done
// for...of preserves input order regardless of delay size
