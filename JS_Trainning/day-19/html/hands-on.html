Task 1: Convert .then to async/await

Rewrite the .then chain below as a single async function showOrders(id) using await and try/catch. Mock both helpers with setTimeout resolving in 300 ms.

// Helpers — mock with setTimeout
function fetchUser(id) {
  return new Promise((resolve) =>
    setTimeout(() => resolve({ id, name: "Priya" }), 300)
  );
}
function fetchOrders(userId) {
  return new Promise((resolve) =>
    setTimeout(() => resolve([{ id: 101 }, { id: 102 }]), 300)
  );
}

// Original .then chain — convert this:
// fetchUser(7)
//   .then(u => fetchOrders(u.id))
//   .then(orders => console.log(orders.length))
//   .catch(e => console.error(e));

// Your async/await rewrite:
async function showOrders(id) {
  try {
    const user   = await fetchUser(id);
    const orders = await fetchOrders(user.id);
    console.log("Order count:", orders.length);   // 2
  } catch (err) {
    console.error("showOrders failed:", err.message);
  }
}

Task 2: Sequential vs Parallel Timing

Write slow() using three separate awaits and fast() using Promise.all. Time both with Date.now() and confirm slow ≈ 1500 ms and fast ≈ 500 ms.

function fetchPrice(id) {
  return new Promise((res) => setTimeout(() => res({ id, price: 100 }), 500));
}

// SLOW — sequential awaits (3 × 500ms = ~1500ms)
async function slow() {
  const t0 = Date.now();
  const a = await fetchPrice(1);
  const b = await fetchPrice(2);
  const c = await fetchPrice(3);
  console.log("slow:", Date.now() - t0, "ms");   // ~1500
}

// FAST — parallel with Promise.all (~500ms)
async function fast() {
  const t0 = Date.now();
  const [a, b, c] = await Promise.all([
    fetchPrice(1),
    fetchPrice(2),
    fetchPrice(3),
  ]);
  console.log("fast:", Date.now() - t0, "ms");   // ~500
}

slow();
fast();

Task 3: Fix the forEach Trap

Run the buggy forEach version and note that "end" appears before any price. Then fix it two ways: for...of and Promise.all + map.

const ids = [1, 2, 3];

function fetchPrice(id) {
  return new Promise((res) => setTimeout(() => res({ id, price: 100 }), 500));
}

// BUG — forEach doesn't await; "end" logs first
async function buggy() {
  console.log("start");
  ids.forEach(async (id) => {
    const p = await fetchPrice(id);
    console.log("got", p);            // ← appears AFTER "end"
  });
  console.log("end");                 // ← appears immediately
}

// FIX 1 — for...of (sequential, ~1500ms)
async function withForOf() {
  console.log("start");
  for (const id of ids) {
    const p = await fetchPrice(id);
    console.log("got", p);
  }
  console.log("end");
}

// FIX 2 — Promise.all + map (parallel, ~500ms)
async function withPromiseAll() {
  console.log("start");
  const results = await Promise.all(ids.map((id) => fetchPrice(id)));
  results.forEach((p) => console.log("got", p));
  console.log("end");
}

buggy();

// withPromiseAll();

Bonus: retry with async/await

Write retry(fn, attempts) that calls async fn() and retries up to attempts times on rejection. After all attempts fail, throw the last error. Test with a flaky() that randomly resolves or rejects.

// Randomly resolves or rejects (~50/50)
function flaky() {
  return new Promise((resolve, reject) => {
    setTimeout(() => {
      if (Math.random() > 0.5) resolve("success!");
      else reject(new Error("random failure"));
    }, 200);
  });
}

async function retry(fn, attempts) {
  let lastErr;
  for (let i = 1; i <= attempts; i++) {
    try {
      const result = await fn();
      console.log(`Succeeded on attempt ${i}`);
      return result;
    } catch (err) {
      lastErr = err;
      console.log(`Attempt ${i} failed: ${err.message}`);
    }
  }
  throw lastErr;   // all attempts exhausted
}

retry(flaky, 5)
  .then((val) => console.log("Final result:", val))
  .catch((err) => console.error("All retries failed:", err.message));
