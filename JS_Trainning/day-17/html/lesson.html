Classes & OOP

ES6 class syntax is syntactic sugar over prototypes — same engine underneath, cleaner surface.

Topic 1: Class Syntax Basics

A class block groups a constructor and methods. The constructor runs when you call new; methods live on the prototype and are shared across all instances.

class User {
  constructor(name, city) {
    this.name = name;   // own property — unique to each instance
    this.city = city;
  }

  greet() {  // lives on User.prototype — shared
    console.log(`Hi, I'm ${this.name} from ${this.city}`);
  }
}

const a = new User("Priya", "Jaipur");
const b = new User("Aarav", "Mumbai");

a.greet();  // "Hi, I'm Priya from Jaipur"
b.greet();  // "Hi, I'm Aarav from Mumbai"

console.log(typeof User);          // "function" — classes ARE functions
console.log(a.greet === b.greet);  // true — same function on the prototype

Topic 2: Getters and Setters

Getters and setters look like property accesses but run code. Use them to derive values or validate on write.

class Product {
  constructor(name, priceInPaise) {
    this.name = name;
    this._priceInPaise = priceInPaise;  // _ = "internal, please don't touch"
  }

  get priceInRupees() {              // called as p.priceInRupees (no parentheses)
    return this._priceInPaise / 100;
  }

  set priceInRupees(rupees) {        // called as p.priceInRupees = 50
    if (rupees < 0) throw new Error("Price cannot be negative");
    this._priceInPaise = rupees * 100;
  }

  get priceWithGST() {               // derived — no backing storage
    return this.priceInRupees * 1.18;
  }
}

const p = new Product("Notebook", 5000);  // ₹50
console.log(p.priceInRupees);   // 50
console.log(p.priceWithGST);    // 59
p.priceInRupees = 100;
console.log(p.priceInRupees);   // 100

Topic 3: extends and super

extends sets up the prototype chain. super() must be called before this in a child constructor — the parent has to initialise this first.

class Animal {
  constructor(name) { this.name = name; }
  speak() { console.log(`${this.name} makes a sound`); }
}

class Dog extends Animal {
  constructor(name, breed) {
    super(name);        // calls Animal's constructor — MUST come first
    this.breed = breed;
  }

  speak() {             // override
    super.speak();      // optionally call the parent version first
    console.log(`${this.name} barks!`);
  }
}

const d = new Dog("Bruno", "Labrador");
d.speak();
// "Bruno makes a sound"
// "Bruno barks!"

console.log(d instanceof Dog);     // true
console.log(d instanceof Animal);  // true

Topic 4: static Methods

Static methods belong to the class itself, not instances. Use them for utilities and factory patterns.

class MathUtils {
  static gst(amount, rate = 18) {
    return amount * (rate / 100);
  }
  static format(amount) {
    return `₹${amount.toFixed(2)}`;
  }
}

console.log(MathUtils.gst(1000));    // 180
console.log(MathUtils.format(1180)); // "₹1180.00"

// Factory pattern — static method that returns a new instance
class User {
  constructor(name) { this.name = name; }
  static fromEmail(email) {
    return new User(email.split("@")[0]);
  }
}

const u = User.fromEmail("priya@example.com");
console.log(u.name);  // "priya"

Topic 5: Private Fields with #

Fields prefixed with # are enforced private by the language — accessing them from outside throws a SyntaxError.

class BankAccount {
  #balance;
  #transactions = [];

  constructor(initial) { this.#balance = initial; }

  deposit(amt) {
    this.#balance += amt;
    this.#transactions.push({ type: "deposit", amt });
  }

  withdraw(amt) {
    if (amt > this.#balance) throw new Error("Insufficient funds");
    this.#balance -= amt;
    this.#transactions.push({ type: "withdraw", amt });
  }

  get balance() { return this.#balance; }
  get history() { return [...this.#transactions]; }
}

const acc = new BankAccount(1000);
acc.deposit(500);
acc.withdraw(200);
console.log(acc.balance);   // 1300
// console.log(acc.#balance); // SyntaxError — enforced by the language

Topic 6: Custom Error Classes

Extend Error to create domain-specific error types you can distinguish with instanceof.

class AppError extends Error {
  constructor(message, code) {
    super(message);
    this.name = this.constructor.name;
    this.code = code;
  }
}

class ValidationError extends AppError {
  constructor(field, message) {
    super(message, "VALIDATION_FAILED");
    this.field = field;
  }
}

function validateAge(age) {
  if (age < 0)   throw new ValidationError("age", "Must be non-negative");
  if (age > 150) throw new ValidationError("age", "Must be under 150");
}

try {
  validateAge(-5);
} catch (e) {
  if (e instanceof ValidationError) {
    console.log(`[${e.code}] ${e.field}: ${e.message}`);
    // "[VALIDATION_FAILED] age: Must be non-negative"
  }
}
