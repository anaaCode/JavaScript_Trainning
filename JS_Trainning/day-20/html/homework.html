Exercise 1: Paper Predictions

Without running the code, write the full output order on paper for all three snippets. Then run each to verify. Annotate which queue each log came from.

// Snippet A
console.log("X");
Promise.resolve().then(() => console.log("Y"));
setTimeout(() => console.log("Z"), 0);
console.log("W");
// Paper prediction: ?   Actual: X, W, Y, Z

// Snippet B
setTimeout(() => console.log("t1"), 0);
setTimeout(() => console.log("t2"), 0);
Promise.resolve().then(() => console.log("p1"));
Promise.resolve().then(() => console.log("p2"));
// Paper prediction: ?   Actual: p1, p2, t1, t2

// Snippet C
async function run() {
  console.log("a");
  await Promise.resolve();
  console.log("b");
}
run();
setTimeout(() => console.log("c"), 0);
console.log("d");
// Paper prediction: ?   Actual: a, d, b, c
// "b" is an await continuation — microtask — beats "c" (macrotask)

Exercise 2: chunk() Helper

Build a chunk(array, fn, chunkSize) helper that processes array in slices of chunkSize, yielding back to the event loop with setTimeout(0) between slices so the page stays responsive. Test with a 100,000-item array.

function chunk(array, fn, chunkSize) {
  function processFrom(index) {
    const end = Math.min(index + chunkSize, array.length);
    for (let i = index; i < end; i++) {
      fn(array[i], i);
    }
    if (end < array.length) {
      setTimeout(() => processFrom(end), 0);   // yield to event loop, then continue
    } else {
      console.log("all done");
    }
  }
  processFrom(0);
}

// Test — 100k items, process 10k at a time
const bigArray = Array.from({ length: 100_000 }, (_, i) => i);
let sum = 0;

chunk(bigArray, (val) => { sum += val; }, 10_000);
// The loop runs in 10 batches, yielding between each.
// The page stays responsive between chunks — try clicking a button while it runs.

Exercise 3: Prove await is a Microtask

Write a snippet that schedules both an await continuation and a setTimeout(0). Predict which runs first, then run to confirm.

async function awaitDemo() {
  await Promise.resolve();          // suspends; continuation queued as microtask
  console.log("await continuation");
}

awaitDemo();
setTimeout(() => console.log("setTimeout"), 0);

// Output:
//   await continuation   ← microtask wins
//   setTimeout           ← macrotask runs after
//
// await suspends the async function and queues its resumption as a microtask.
// setTimeout queues its callback as a macrotask.
// Microtask queue drains before the macrotask queue gets a turn.

Exercise 4: X, W, Y, Z Deep Dive

Take this snippet and draw the exact state of the call stack, microtask queue, and macrotask queue after each statement executes. Write it as a comment table.

console.log("X");
Promise.resolve().then(() => console.log("Y"));
setTimeout(() => console.log("Z"), 0);
console.log("W");

// Step-by-step queue table:
//
// After console.log("X"):
//   Stack:     empty
//   Microtask: []
//   Macrotask: []
//   Output so far: X
//
// After Promise.resolve().then(...):
//   Stack:     empty
//   Microtask: [log Y]
//   Macrotask: []
//   Output so far: X
//
// After setTimeout(...):
//   Stack:     empty
//   Microtask: [log Y]
//   Macrotask: [log Z]
//   Output so far: X
//
// After console.log("W"):
//   Stack:     empty  ← sync phase complete
//   Microtask: [log Y]
//   Macrotask: [log Z]
//   Output so far: X, W
//
// Event loop — drain microtasks:
//   Run log Y → output: X, W, Y
//   Microtask: []
//
// Event loop — one macrotask:
//   Run log Z → output: X, W, Y, Z
//
// Final output: X, W, Y, Z
