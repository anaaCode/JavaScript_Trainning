Exercise 1: compose from Scratch

Implement compose(...fns) (right-to-left). Use it with the same addOne, square, negate from the hands-on. Confirm the output ordering differs from pipe and explain why.

const compose = (...fns) => (x) => fns.reduceRight((acc, fn) => fn(acc), x);

// compose(addOne, square, negate)(5)
//   negate(5) → -5 → square(-5) → 25 → addOne(25) → 26
const withCompose = compose(addOne, square, negate);
console.log(withCompose(5));   // 26  (vs pipe which gives -36)

                


                

                    
Exercise 2: Refactor a Mutating Function

                    
Take any function from a previous exercise that mutates an array. Rewrite it as a pure function. Verify the original array is untouched after calling your pure version.

                    
// IMPURE — mutates original
function addTagImpure(tags, newTag) {
  tags.push(newTag);
  return tags;
}

// PURE — spreads into a new array
function addTagPure(tags, newTag) {
  return [...tags, newTag];
}

const original = ["js", "css"];
const updated  = addTagPure(original, "html");

console.log("original →", original);  // ["js", "css"]  ← unchanged ✓
console.log("updated  →", updated);   // ["js", "css", "html"]

                    

                


                

                    
Exercise 3: Generalised curry(fn)

                    
Implement curry(fn): given a function of N arguments, return a curried version that collects arguments one call at a time until it has enough to call fn. Test it with a 3-argument add.

                    
function curry(fn) {
  return function curried(...args) {
    if (args.length >= fn.length) return fn.apply(this, args);
    return (...more) => curried(...args, ...more);
  };
}

function add(a, b, c) { return a + b + c; }

const curriedAdd = curry(add);
console.log(curriedAdd(1)(2)(3));    // 6
console.log(curriedAdd(1, 2)(3));    // 6
console.log(curriedAdd(1)(2, 3));    // 6
console.log(curriedAdd(1, 2, 3));    // 6

                    

                


                

                    
Exercise 4: Average Order Value Pipeline

                    
Take the order data from Topic 7 and write a pipeline that returns the average order value (excluding GST). Use small pure functions only — no logic inside the pipe call itself.

                    
const orders = [
  { id: 1, item: "Pen",  price: 50,  quantity: 2 },
  { id: 2, item: "Book", price: 200, quantity: 1 },
  { id: 3, item: "Bag",  price: 800, quantity: 1 },
  { id: 4, item: "Mug",  price: 150, quantity: 3 },
];

const toLineTotals = (arr) => arr.map((o) => o.price * o.quantity);
const average      = (arr) => arr.reduce((a, b) => a + b, 0) / arr.length;
const round2       = (n)   => Math.round(n * 100) / 100;
const formatINR    = (n)   => `₹${n.toLocaleString("en-IN")}`;

const avgOrderValue = pipe(toLineTotals, average, round2, formatINR);
console.log(avgOrderValue(orders));   // ₹387.5
