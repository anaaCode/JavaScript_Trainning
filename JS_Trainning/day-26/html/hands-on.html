Task 1: Tiny EventEmitter

What to do: build createEmitter() returning { on, off, emit }.

  • Register two listeners on event "hello". Emit it — both should fire.
  • Call off on one listener. Emit again — only one should fire.
function createEmitter() {
  const listeners = new Map();

  return {
    on(event, callback) {
      if (!listeners.has(event)) listeners.set(event, new Set());
      listeners.get(event).add(callback);
    },
    off(event, callback) {
      listeners.get(event)?.delete(callback);
    },
    emit(event, ...args) {
      listeners.get(event)?.forEach((cb) => cb(...args));
    },
  };
}

const bus = createEmitter();
const greetA = () => console.log("listener A fired");
const greetB = () => console.log("listener B fired");

bus.on("hello", greetA);
bus.on("hello", greetB);
bus.emit("hello");     // listener A fired  /  listener B fired

bus.off("hello", greetA);
bus.emit("hello");     // listener B fired

                


                

                    
Task 2: User Factory

                    
What to do: write a createUser(name, role) factory. Default role to "user". The returned object must have name, role, and a method canEdit() that returns true only when role === "admin".

                    
function createUser(name, role = "user") {
  return {
    name,
    role,
    canEdit() { return this.role === "admin"; },
  };
}

const u1 = createUser("Priya");
const u2 = createUser("Aarav", "admin");

console.log(u1.name, u1.canEdit());   // Priya false
console.log(u2.name, u2.canEdit());   // Aarav true

                    

                


                

                    
Task 3: Module-Singleton Cache

                    
What to do: build a createCache() factory that returns { get, set, has, size } with private Map storage.

                    
    
                        
  • Show that two separate calls to createCache() give two independent caches.
    • 
                        
  • Explain (in a comment) how exporting default createCache() from a module gives a single shared cache.
    • 
                    

                    
function createCache() {
  const store = new Map();   // private

  return {
    set(key, value) { store.set(key, value); },
    get(key)        { return store.get(key); },
    has(key)        { return store.has(key); },
    get size()      { return store.size; },
  };
}

const cacheA = createCache();
const cacheB = createCache();

cacheA.set("user", "Priya");
console.log(cacheA.get("user"));   // Priya
console.log(cacheB.has("user"));   // false — independent
console.log(cacheA.size);          // 1
console.log(cacheB.size);          // 0

// In a module file: export default createCache();
// → every importer receives the SAME instance (module evaluated once)

                    

                


                

                    
Bonus: Reactive Counter with Proxy

                    
What to do: build reactive(obj, onChange) that wraps obj in a Proxy and calls onChange(prop, value) on every set. Wrap { count: 0 } and confirm that every assignment logs "count → X". Try setting a non-existent property — confirm the trap fires for that too.

                    
function reactive(obj, onChange) {
  return new Proxy(obj, {
    set(target, prop, value) {
      target[prop] = value;
      onChange(prop, value);
      return true;
    },
  });
}

const state = reactive({ count: 0 }, (key, val) =>
  console.log(`${key} → ${val}`)
);

state.count = 1;      // count → 1
state.count = 5;      // count → 5
state.name  = "Riya"; // name → Riya  (new property — trap fires too)
