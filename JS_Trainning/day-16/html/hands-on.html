Task 1: Inspect a Prototype

What to do:

  • Create const arr = [1, 2, 3];
  • Use Object.getPrototypeOf(arr) and log it — what is it?
  • Log one more step — what is the prototype of that? And one more — what value ends the chain?
  • In a comment, draw the chain.
const arr = [1, 2, 3];

// Step 1 — prototype of arr
console.log(Object.getPrototypeOf(arr));  // Array.prototype (has push, map…)

// Step 2 — prototype of Array.prototype
console.log(Object.getPrototypeOf(Object.getPrototypeOf(arr)));  // Object.prototype

// Step 3 — end of chain
console.log(Object.getPrototypeOf(Object.getPrototypeOf(Object.getPrototypeOf(arr))));  // null

// Chain:  arr  →  Array.prototype  →  Object.prototype  →  null

Task 2: Build with Object.create

What to do:

  • Create an object vehicle with a method start() { console.log(`${this.name} starting`); }
  • Use Object.create(vehicle) to make car, set car.name = "Tata Nexon"
  • Make bike the same way with name = "Royal Enfield"
  • Call start() on both
  • Use hasOwnProperty and in to confirm name is own and start is inherited
const vehicle = {
  start() { console.log(`${this.name} starting`); },
};

const car  = Object.create(vehicle);
car.name   = "Tata Nexon";

const bike = Object.create(vehicle);
bike.name  = "Royal Enfield";

car.start();   // "Tata Nexon starting"
bike.start();  // "Royal Enfield starting"

console.log(car.hasOwnProperty("name"));   // true  ← own
console.log(car.hasOwnProperty("start"));  // false ← inherited
console.log("name"  in car);               // true  ← found on car
console.log("start" in car);               // true  ← found on vehicle (chain)

Task 3: Constructor Function Inheritance

What to do:

  • Build a Person(name) constructor that sets this.name
  • Add Person.prototype.greet that logs "Hi, I'm " + this.name
  • Build a Student(name, school) constructor that calls Person.call(this, name) and sets this.school
  • Link Student.prototype to inherit from Person.prototype using Object.create
  • Add Student.prototype.study that logs the student's name and school
  • Create new Student("Riya", "IIT Delhi") and call both greet and study
function Person(name) {
  this.name = name;
}

Person.prototype.greet = function () {
  console.log("Hi, I'm " + this.name);
};

function Student(name, school) {
  Person.call(this, name);  // borrow Person's constructor
  this.school = school;
}

Student.prototype = Object.create(Person.prototype);
Student.prototype.constructor = Student;

Student.prototype.study = function () {
  console.log(this.name + " studies at " + this.school);
};

const riya = new Student("Riya", "IIT Delhi");
riya.greet();  // "Hi, I'm Riya"
riya.study();  // "Riya studies at IIT Delhi"

console.log(Object.getPrototypeOf(riya) === Student.prototype);             // true
console.log(Object.getPrototypeOf(Student.prototype) === Person.prototype); // true

Bonus: hasOwnProperty vs in

What to do:

  • Create const dog = Object.create({ species: "Canis" }); dog.name = "Bruno";
  • Predict the result of each check below, then run all five and match your predictions
  • In a comment, write the one-line rule for when to use which
const dog = Object.create({ species: "Canis" });
dog.name = "Bruno";

console.log(dog.hasOwnProperty("name"));     // true   — name is own
console.log(dog.hasOwnProperty("species"));  // false  — species is inherited
console.log("name" in dog);                  // true   — found on dog
console.log("species" in dog);               // true   — found on prototype
console.log("toString" in dog);              // true   — found on Object.prototype

// Rule: use hasOwnProperty to check ONLY direct properties;
//       use 'in' when the whole chain (including inherited) should be checked.
