Exercise 1: 3-Level Object.create Chain

What to do:

  • Use Object.create to build a 3-level chain: tool → vehicle → car
  • Each level adds one method: tool.describe, vehicle.drive, car.honk
  • Create a final object (e.g. myCar) inheriting from car and confirm all three methods are reachable
const tool = {
  describe() { console.log(`${this.name} is a tool`); },
};

const vehicle = Object.create(tool);
vehicle.drive = function () { console.log(`${this.name} is driving`); };

const car = Object.create(vehicle);
car.honk = function () { console.log(`${this.name} honks!`); };

const myCar = Object.create(car);
myCar.name = "Tata Nexon";

myCar.describe();  // "Tata Nexon is a tool"    ← from tool
myCar.drive();     // "Tata Nexon is driving"    ← from vehicle
myCar.honk();      // "Tata Nexon honks!"        ← from car

// Chain: myCar → car → vehicle → tool → Object.prototype → null

Exercise 2: Shape → Circle Inheritance

What to do:

  • Build a Shape(name) constructor with a describe() method on the prototype
  • Build Circle(name, radius) that inherits from Shape and adds an area() method
  • Test with a Circle of name "C1" and radius 5
  • Call both describe() and area(), then verify the prototype chain
function Shape(name) {
  this.name = name;
}

Shape.prototype.describe = function () {
  console.log(`I am a shape called ${this.name}`);
};

function Circle(name, radius) {
  Shape.call(this, name);   // inherit own properties
  this.radius = radius;
}

Circle.prototype = Object.create(Shape.prototype);
Circle.prototype.constructor = Circle;

Circle.prototype.area = function () {
  const a = Math.PI * this.radius * this.radius;
  console.log(`Area of ${this.name}: ${a.toFixed(2)}`);
};

const c1 = new Circle("C1", 5);
c1.describe();  // "I am a shape called C1"
c1.area();      // "Area of C1: 78.54"

console.log(Object.getPrototypeOf(c1) === Circle.prototype);             // true
console.log(Object.getPrototypeOf(Circle.prototype) === Shape.prototype); // true

Exercise 3: Inspect a Built-in Prototype

What to do:

  • Open the console and inspect String.prototype (or any built-in)
  • Find five methods you didn't know about
  • Document one with a short code example below
// Five lesser-known String.prototype methods:

// 1. padStart — pad the beginning to reach a target length
console.log("5".padStart(3, "0"));    // "005"

// 2. padEnd — pad the end instead
console.log("hi".padEnd(5, "."));     // "hi..."

// 3. repeat — repeats the string n times
console.log("ab".repeat(3));          // "ababab"

// 4. at — supports negative indices (unlike bracket notation)
console.log("hello".at(-1));          // "o"

// 5. trimStart — removes leading whitespace only
console.log("   hello".trimStart());  // "hello"

Exercise 4: chainOf(obj)

What to do:

  • Write a function chainOf(obj) that returns an array of every prototype in obj's chain, ending with null
  • Test it on a plain object, an array, and one of your constructor instances
function chainOf(obj) {
  const chain = [];
  let current = Object.getPrototypeOf(obj);
  while (current !== null) {
    chain.push(current);
    current = Object.getPrototypeOf(current);
  }
  chain.push(null);
  return chain;
}

// Plain nested Object.create chain
const base   = { base: true };
const mid    = Object.create(base);
const top    = Object.create(mid);
console.log(chainOf(top));
// [ base, Object.prototype, null ]

// Array
console.log(chainOf([]));
// [ Array.prototype, Object.prototype, null ]

// Constructor instance (re-use Circle from Exercise 2)
const c2 = new Circle("C2", 3);
console.log(chainOf(c2));
// [ Circle.prototype, Shape.prototype, Object.prototype, null ]
