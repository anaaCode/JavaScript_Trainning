Exercise 1: Barrel for Three Utility Modules

Create js/homework-barrel/ with three modules: stringUtils.js, arrayUtils.js, and mathUtils.js. Write an index.js barrel that re-exports all of them using both export * forms. Import everything in homework.js from the barrel in a single statement.

// homework-barrel/stringUtils.js
export function capitalize(str) { ... }
export function truncate(str, max) { ... }
export function isPalindrome(str) { ... }

// homework-barrel/arrayUtils.js
export function unique(arr) { ... }
export function flatten(arr) { ... }
export function groupBy(arr, key) { ... }

// homework-barrel/mathUtils.js
export function clamp(value, min, max) { ... }
export function lerp(a, b, t) { ... }
export function roundTo(n, decimals) { ... }

// homework-barrel/index.js — barrel with export *
export * from "./stringUtils.js";
export * from "./arrayUtils.js";
export * from "./mathUtils.js";

// homework.js — single import from the barrel
import { capitalize, unique, clamp, ... } from "./homework-barrel/index.js";
console.log(capitalize("hello"));    // Hello
console.log(unique([1,2,2,3,3,3])); // [1, 2, 3]
console.log(clamp(150, 0, 100));    // 100

Open Console to verify all 9 utility functions log correct output.

Exercise 2: Dynamic import() for a Lazy Feature

Reuse heavy.js from hands-on. Write a loadHeavy() async function in homework.js that uses await import("./heavy.js") to load compute() on demand. Wire the button below to trigger it — watch DevTools Network to confirm the file is fetched only on click (requires a local HTTP server).

async function loadHeavy() {
  const { compute, heavyTask } = await import("./heavy.js");
  console.log("compute()   →", compute());    // "computed!"
  console.log("heavyTask() →", heavyTask());
}

loadHeavy();

Exercise 3: Singleton Store Shared Across Services

Create homework-cache.js exporting a single shared Map as default. Create homework-userService.js and homework-productService.js — each imports the store and writes to it. In homework.js, import the store directly and confirm both services' writes are visible from the same Map instance.

// homework-cache.js
const store = new Map();
export default store;

// homework-userService.js
import store from "./homework-cache.js";
export function saveUser(id, data) { store.set(`user:${id}`, data); }

// homework-productService.js
import store from "./homework-cache.js";
export function saveProduct(id, data) { store.set(`product:${id}`, data); }

// homework.js
import store from "./homework-cache.js";
import { saveUser }    from "./homework-userService.js";
import { saveProduct } from "./homework-productService.js";

saveUser(1, { name: "Arjun" });
saveProduct(10, { name: "Notebook" });

console.log([...store.entries()]);
// [["user:1", { name:"Arjun" }], ["product:10", { name:"Notebook" }]]

                


                

                    
Exercise 4: Build a Circular Dependency — Observe & Fix

                    
Manually create a.js and b.js that import from each other and read a top-level value. Observe the undefined bug in the Console. Then fix it by either:

                    
    
                        
  • Extracting the shared value to a shared.js module both files import from, or
    • 
                        
  • Moving the read inside a function (lazy access) so it runs after both modules finish evaluating
    • 
                    

                    
// a.js — broken circular dep
import { fromB } from "./b.js";
export const fromA = "I am A";
console.log("a sees fromB =", fromB);   // undefined!

// b.js — broken circular dep
import { fromA } from "./a.js";
export const fromB = "I am B";
console.log("b sees fromA =", fromA);   // undefined!

// ─── Fix 1: shared.js ────────────────────────
// shared.js
export const MESSAGE = "shared value";

// a.js  →  import { MESSAGE } from "./shared.js";
// b.js  →  import { MESSAGE } from "./shared.js";

// ─── Fix 2: lazy access ──────────────────────
// a.js
import { fromB } from "./b.js";
export const fromA = "I am A";
export function getFromB() { return fromB; }   // read inside function ✓

                    
This exercise is intentionally done in your own editor — create the files, run them, observe the bug, then apply the fix.
